รูปแบบเดิม (หน่วงความเร็วเว็บ)

  • เรียกใช้คำสั่ง `$conn->query($sql)` ฝังอยู่กลางโค้ด HTML ของหน้าเว็บ
  • เบราว์เซอร์ถูกสั่งให้รอให้ฝั่งเซิร์ฟเวอร์รัน SQL และประมวลผล PHP ลูปจนจบก่อนส่งข้อมูล
  • คิวรีหนักๆ หรือมีปัญหา Time-out จะทำให้เว็บค้างขาวทั้งหน้าจอ ใช้งานส่วนอื่นไม่ได้

รูปแบบใหม่ (ทำงานเบื้องหลัง)

  • โหลด Layout หลักของเว็บเสร็จใน 0.1 วินาทีแรก เมนู แผงข้าง พร้อมให้กดสั่งงานทันที
  • ใช้ Javascript ส่ง Request ไปหลังบ้านแบบ Asynchronous ด้วย `fetch()` API
  • มี Loading Skeleton สวยงามแสดงทดแทนระหว่างรอคอยผลลัพธ์ข้อมูลจาก API

3. รูปแบบการปรับโครงสร้างโค้ด

ตัวอย่างการแยกคิวรีข้อมูลหนักๆ ไปทำในสเปค API และใช้ Javascript เรียกข้อมูลจากหลังบ้าน

1. สร้างไฟล์ API หลังบ้าน (PHP) » api/get_daily_sales_json.php 
// ----------------------------------------------------
// ไฟล์คืนข้อมูลเป็น JSON เท่านั้น เพื่อส่งกลับไปให้เบราว์เซอร์เอาไปวาดผล
// ----------------------------------------------------
header('Content-Type: application/json');
include('../config/connect_db.php');

try {
    $date = date("d/m/Y");
    $sql = "SELECT BRANCH, SUM(CAST(TRD_G_KEYIN AS DECIMAL(10,2))) as TOTAL_AMT 
            FROM ims_product_sale_cockpit 
            WHERE DI_DATE = :di_date 
            GROUP BY BRANCH";
            
    $stmt = $conn->prepare($sql);
    $stmt->execute([':di_date' => $date]);
    $results = $stmt->fetchAll(PDO::FETCH_ASSOC);
    
    // ส่งข้อมูลกลับไปแบบ JSON อย่างรวดเร็ว
    echo json_encode([
        'success' => true,
        'data' => $results
    ]);
} catch (Exception $e) {
    echo json_encode(['success' => false, 'message' => $e->getMessage()]);
}


2. เขียนดึงข้อมูลฝั่งหน้าจอ (JS) » Dashboard_sale.php 
// ----------------------------------------------------
// ใช้ Fetch API ยิงขอข้อมูลเบื้องหลังหลังจากโครงสร้างเว็บแสดงขึ้นมาแล้ว
// ----------------------------------------------------
document.addEventListener("DOMContentLoaded", function() {
    // โทรไปหา API หลังบ้าน
    fetch('api/get_daily_sales_json.php')
        .then(response => response.json())
        .then(res => {
            if (res.success) {
                const tbody = document.getElementById('tableDailyBody');
                tbody.innerHTML = ''; 
                
                res.data.forEach(row => {
                    const tr = document.createElement('tr');
                    tr.innerHTML = `<td>\${row.BRANCH}</td>
                                    <td>\${parseFloat(row.TOTAL_AMT).toLocaleString()}</td>`;
                    tbody.appendChild(tr);
                });
                
                // ซ่อนตัวหมุน Loading และแสดงผลข้อมูลจริง
                document.getElementById('table-loader').classList.add('d-none');
                document.getElementById('tableDaily').classList.remove('d-none');
            }
        });
});
