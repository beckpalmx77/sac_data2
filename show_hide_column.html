คำแนะนำการเพิ่มปุ่ม Checkbox / ตัวเลือกสำหรับแสดงและซ่อน Column ใน DataTables

สำหรับหน้าจอ forecast_plan_display_realtime.php

ระบบจำการตั้งค่าคอลัมน์แบบ Hybrid (LocalStorage + DB MySQL)

ระบบทำงานร่วมกันอย่างสมบูรณ์แบบ 2 ระดับ: โหลดจาก LocalStorage ทันทีเพื่อให้แสดงผลรวดเร็วแบบ 0ms latency และซิงก์ข้อมูลกับ Database MySQL (ตาราง ims_user_column_config) ในเบื้องหลัง เพื่อให้ผู้ใช้แต่ละท่านเมื่อไปเข้าใช้งานที่เครื่องอื่น (มือถือ, แท็บเล็ต, PC เครื่องอื่น) ก็ยังได้การตั้งค่าคอลัมน์เดิมเสมอ

2 แนวทางการ 구현 (Implementation Methods)

แนวทางที่ 1: Custom UI Checkbox Modal (แนะนำที่สุด ⭐⭐⭐⭐⭐)

สร้างปุ่ม Modal รวมรายการ Checkbox ของแต่ละ Column สามารถแบ่งกลุ่มคอลัมน์ให้เลือกปิด/เปิดได้ง่าย เช่น กลุ่มข้อมูลสินค้า, กลุ่มยอดขายย้อนหลัง, กลุ่มเป้าหมาย/แผนงาน, กลุ่มพื้นที่

  • ข้อดี: สวยงาม ใช้งานง่าย จัดหมวดหมู่คอลัมน์ได้ดี
  • ฟีเจอร์พิเศษ: บันทึกสถานะการเปิด/ปิดลง localStorage ทำให้เมื่อรีเฟรชหน้าเว็บ ค่าที่เคยซ่อน/แสดงยังคงอยู่เหมือนเดิม
แนวทางที่ 2: DataTables ColVis Button (Built-in)

ใช้งาน DataTables Buttons Extension (colvis) ซึ่งเป็นปลั๊กอินมาตรฐานของ DataTables เพียงเรียกใช้ CSS/JS ของ Buttons และตั้งค่าใน DataTables Options

  • ข้อดี: โค้ดน้อย ตั้งค่าง่ายมาก รวดเร็ว
  • ข้อจำกัด: แสดงเป็นเมนูดรอปดาวน์รายการยาวๆ ไม่สามารถแบ่งกลุ่มหมวดหมู่ย่อยได้สวยงามเท่า Modal Custom

ขั้นตอนที่ 1.3: เพิ่ม JavaScript ในส่วนการทำงานของ DataTables

ในไฟล์ forecast_plan_display_realtime.php บริเวณด้านล่างต่อจาก const table = $('#planTable').DataTable({...}); ให้ใส่สคริปต์นี้เข้าไป:

// ========================================================
// ระบบจัดการ ซ่อน / แสดง Column DataTables + LocalStorage (แยกราย User)
// ========================================================
const currentUser = ;
const STORAGE_KEY = 'forecast_realtime_columns_visible_' + currentUser;

// ฟังก์ชันโหลดค่าสถานะที่บันทึกไว้ใน LocalStorage
function loadColumnVisibility() {
    const savedState = localStorage.getItem(STORAGE_KEY);
    if (!savedState) return;

    try {
        const visibleCols = JSON.parse(savedState);
        // วนลูปตั้งค่า Column ตามค่าที่เคยบันทึกไว้
        $.each(visibleCols, function(colIdx, isVisible) {
            const column = table.column(colIdx);
            if (column.length && colIdx != 23 && colIdx != 1) { // ยกเว้นคอลัมน์สำคัญที่ตรึงไว้
                column.visible(isVisible, false);
                $(`#col_${colIdx}`).prop('checked', isVisible);
            }
        });
        table.columns.adjust().draw(false);
    } catch (e) {
        console.error("Error loading column visibility settings", e);
    }
}

// เมื่อมีการกดเปลี่ยนแปลง Checkbox
$('.col-toggle-chk').on('change', function () {
    const colIdx = $(this).data('column');
    const isChecked = $(this).is(':checked');

    // สั่ง DataTables ให้แสดง/ซ่อนคอลัมน์นั้นๆ
    const column = table.column(colIdx);
    column.visible(isChecked);

    // บันทึกสถานะใหม่ลง LocalStorage
    saveColumnVisibility();
});

// ฟังก์ชันบันทึกค่าสถานะ Checkbox ทั้งหมดลง LocalStorage
function saveColumnVisibility() {
    const colState = {};
    $('.col-toggle-chk').each(function() {
        const colIdx = $(this).data('column');
        colState[colIdx] = $(this).is(':checked');
    });
    localStorage.setItem(STORAGE_KEY, JSON.stringify(colState));
}

// ปุ่มเลือกทั้งหมด
$('#btnSelectAllCols').on('click', function() {
    $('.col-toggle-chk').not(':disabled').prop('checked', true).trigger('change');
});

// ปุ่มคืนค่าเริ่มต้น (แสดงทุกคอลัมน์)
$('#btnResetCols').on('click', function() {
    localStorage.removeItem(STORAGE_KEY);
    $('.col-toggle-chk').not(':disabled').prop('checked', true);
    for (let i = 0; i < 24; i++) {
        table.column(i).visible(true, false);
    }
    table.columns.adjust().draw(false);
});

// เรียกใช้งานฟังก์ชันโหลดตั้งค่าเมื่อ DataTables สร้างเสร็จแล้ว
loadColumnVisibility();

ตารางอ้างอิงลำดับ Index ของ คอลัมน์ (Column Index Reference)

รายละเอียด Index ของคอลัมน์ทั้ง 24 คอลัมน์ ใน forecast_plan_display_realtime.php:

Index ชื่อ Data Key หัวข้อคอลัมน์ (Header Title) หมวดหมู่ (Category) การซ่อน/แสดง
0 ar_code รหัสร้านค้า ร้านค้า/สินค้า ซ่อน/แสดง ได้
1 ar_name ชื่อร้านค้า ร้านค้า/สินค้า ตรึงไว้ (แนะนำให้แสดง)
2 brand_code ยี่ห้อ ร้านค้า/สินค้า ซ่อน/แสดง ได้
3 sku_code รหัสสินค้า (SKU) ร้านค้า/สินค้า ซ่อน/แสดง ได้
4 avg_per_month_year เฉลี่ย 12ด. สถิติย้อนหลัง ซ่อน/แสดง ได้
5 max_qty_year MAX (ปีก่อนหน้า) สถิติย้อนหลัง ซ่อน/แสดง ได้
6 min_qty_year MIN (ปีก่อนหน้า) สถิติย้อนหลัง ซ่อน/แสดง ได้

คำแนะนำเพิ่มเติมสำหรับการใช้งาน FixedColumns

เนื่องจากตาราง #planTable มีการใช้งาน fixedColumns: { left: 4 } (ตรึง 4 คอลัมน์แรก) หากผู้ใช้งานทำการซ่อนคอลัมน์ใน 4 คอลัมน์แรก ควรเรียกฟังก์ชัน table.columns.adjust().draw(false); เพื่อให้ DataTables คำนวณความกว้างและตำแหน่งการตรึงคอลัมน์ใหม่ได้อย่างถูกต้องครับ
