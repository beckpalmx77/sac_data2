ผลการวิเคราะห์ภาพรวมความมั่นคงปลอดภัย

จากการสแกนและตรวจสอบ Source Code ภาษา PHP ในระบบ sac_data2 จำนวนทั้งหมด 1,020 ไฟล์ พบไฟล์ที่มีช่องโหว่ความเสี่ยง SQL Injection ทั้งสิ้น 293 ไฟล์ คิดเป็น 28.7% ของระบบ และพบจุดบกพร่องรวม 1,706 จุด

รูปแบบช่องโหว่หลักที่พบในระบบ (Vulnerability Patterns & Code Diffs)
ต่อ String ตัวแปร User Input ตรงๆ ใน SQL
Critical (1,024 จุด)
นำตัวแปรจาก $_POST หรือ $_GET มาเชื่อมต่อลงในคำสั่ง SELECT, INSERT, UPDATE, DELETE โดยตรง โดยไม่มีการ Escape หรือ Bind Parameter
ตัวอย่างโค้ดเดิมที่มีช่องโหว่ (Vulnerable Code)
// ไฟล์ model/manage_account_process.php
$id = $_POST["id"];
$sql_get = "SELECT * FROM ims_user WHERE id = " . $id;
$statement = $conn->query($sql_get);

// หรือการค้นหา
$sql_find = "SELECT * FROM ims_user WHERE email = '" . $email . "'";
$nRows = $conn->query($sql_find)->fetchColumn();
วิธีแก้ไขที่ถูกต้อง (PDO Prepared Statements)
// ใช้ PDO Prepare + Bind Parameter
$stmt = $conn->prepare("SELECT * FROM ims_user WHERE id = :id");
$stmt->execute([':id' => $id]);
$results = $stmt->fetchAll(PDO::FETCH_ASSOC);

// หรือการตรวจสอบแถว
$stmt = $conn->prepare("SELECT COUNT(*) FROM ims_user WHERE email = :email");
$stmt->execute([':email' => $email]);
$nRows = $stmt->fetchColumn();
DataTables Column / Order By Injection
High (152 จุด)
รับค่า $_POST['columns'][$col]['data'] และ $_POST['order'][0]['dir'] จากฝั่ง Client แล้วนำมาต่อเข้า ORDER BY โดยตรง ทำให้เกิด 2nd Order / Blind SQL Injection
ตัวอย่างโค้ดเดิมที่มีช่องโหว่ (Vulnerable Code)
$columnName = $_POST['columns'][$columnIndex]['data'];
$columnSortOrder = $_POST['order'][0]['dir'];

$stmt = $conn->prepare("SELECT * FROM ims_user WHERE 1 " . $searchQuery
    . " ORDER BY " . $columnName . " " . $columnSortOrder . " LIMIT :limit,:offset");
วิธีแก้ไขที่ถูกต้อง (Whitelist Validation)
// ทำ Whitelist คอลัมน์ที่อนุญาตให้เรียงลำดับเท่านั้น
$allowedColumns = ['id' => 'id', 'email' => 'email', 'first_name' => 'first_name', 'status' => 'status'];
$sortColumn = $allowedColumns[$columnName] ?? 'id';
$sortDir = strtolower($columnSortOrder) === 'desc' ? 'DESC' : 'ASC';

$stmt = $conn->prepare("SELECT * FROM ims_user WHERE 1 " . $searchQuery
    . " ORDER BY {$sortColumn} {$sortDir} LIMIT :limit, :offset");
Fake / Ineffective prepare() Calls
High (43 จุด)
เรียกใช้ฟังก์ชัน $conn->prepare() แต่ภายในกลับนำตัวแปรมาต่อ String เช่นเดิม ทำให้ไม่มีประโยชน์ในการป้องกัน SQL Injection
ตัวอย่างโค้ดเดิมที่มีช่องโหว่ (Vulnerable Code)
// ไฟล์ model/customer_ajaxfile.php
$stmt = $conn->prepare("SELECT * FROM v_customer_salename WHERE SLMN_SLT LIKE " . $manage_team_id . " GROUP BY AR_CODE LIMIT :limit");

// หรือไฟล์ลบข้อมูล
$sql = "DELETE FROM ims_user WHERE id = " . $id;
$query = $conn->prepare($sql);
$query->execute();
วิธีแก้ไขที่ถูกต้อง (Parameterized Query)
// ใช้ Placeholder (?) หรือ (:named) เสมอ
$stmt = $conn->prepare("SELECT * FROM v_customer_salename WHERE SLMN_SLT LIKE :team_id GROUP BY AR_CODE LIMIT :limit");
$stmt->bindValue(':team_id', '%' . $manage_team_id . '%', PDO::PARAM_STR);
$stmt->bindValue(':limit', (int)$numberofrecords, PDO::PARAM_INT);
$stmt->execute();

// ลบข้อมูล
$stmt = $conn->prepare("DELETE FROM ims_user WHERE id = :id");
$stmt->execute([':id' => $id]);
REST API & Export Query Execution
Critical (477 จุด)
ไฟล์ในโฟลเดอร์ api/, engine/, export_process/ นำ $_GET['id'] หรือตัวกรองวันที่/ลูกค้า มาประกอบ Query แล้วเรียก $conn->query($sql)
ตัวอย่างโค้ดเดิมที่มีช่องโหว่ (Vulnerable Code)
// ไฟล์ api/customer_data_api.php
if (isset($_GET['id']) && !empty($_GET['id'])) {
    $id = $_GET['id'];
    $sql_get = "SELECT * FROM ims_provinces WHERE id = " . $id;
}
$statement = $conn->query($sql_get);
วิธีแก้ไขที่ถูกต้อง (API Parameterized Prepared Query)
if (isset($_GET['id']) && !empty($_GET['id'])) {
    $stmt = $conn->prepare("SELECT * FROM ims_provinces WHERE id = :id");
    $stmt->execute([':id' => (int)$_GET['id']]);
} else {
    $stmt = $conn->prepare("SELECT * FROM ims_provinces");
    $stmt->execute();
}
$results = $stmt->fetchAll(PDO::FETCH_ASSOC);

    แก้ไขไฟล์กลุ่มเสี่ยงวิกฤตสูงสุดทันที (Priority 1: REST API & Authentication & Management Process)

    เริ่มต้นแก้ไขไฟล์ในโฟลเดอร์ api/, model/manage_*_process.php และ engine/ ที่รับข้อมูลจาก $_POST และ $_GET โดยแปลงคำสั่ง SQL ทั้งหมดให้เป็น PDO Prepared Statements โดยห้ามมีเครื่องหมายเชื่อมต่อ String (.) ของตัวแปรภายนอกใน SQL เด็ดขาด

  • 2

    สร้าง Standard Helper Function สำหรับ DataTables Server-Side Processing

    สร้างฟังก์ชันกลางสำหรับ Validate คอลัมน์ที่ส่งมาจาก DataTables เพื่อป้องกัน ORDER BY Injection และลดการเขียนโค้ดซ้ำซ้อนในไฟล์กว่า 70 ไฟล์ใน model/ 

    // ตัวอย่างฟังก์ชันกลาง util/datatables_helper.php
function getSafeOrderClause(array $allowedColumns, $requestedColIdx, $requestedDir, $defaultCol = 'id') {
    $colName = $allowedColumns[$requestedColIdx] ?? $defaultCol;
    $dir = strtoupper($requestedDir) === 'DESC' ? 'DESC' : 'ASC';
    return " ORDER BY `{$colName}` {$dir} ";
}

    ตั้งค่า Database Connection Options ให้ปลอดภัย (Disable Emulated Prepares)

    ในไฟล์ config/connect_db.php ให้กำหนด PDO::ATTR_EMULATE_PREPARES => false เพื่อบังคับให้ MySQL Server ทำการ Prepare คำสั่งจริง ซึ่งจะป้องกัน Second-Order และ Multi-statement Injection ได้อย่างสมบูรณ์ 

    $conn = new PDO(
    "mysql:host=".DB_HOST.";dbname=".DB_NAME.";port=".DB_PORT.";charset=utf8mb4",
    DB_USER, 
    DB_PASS,
    [
        PDO::ATTR_ERRMODE => PDO::ERRMODE_EXCEPTION,
        PDO::ATTR_DEFAULT_FETCH_MODE => PDO::FETCH_ASSOC,
        PDO::ATTR_EMULATE_PREPARES => false // ปิดการจำลอง Prepare เพื่อความปลอดภัยสูงสุด
    ]
);
